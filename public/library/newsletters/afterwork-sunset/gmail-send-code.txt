// Leave RECIPIENT blank to create a review draft addressed to your own Gmail.
// Enter one recipient address here before using sendYsabelNewsletter.
const RECIPIENT = '';
const NEWSLETTER_URL = 'https://ysabelsociety.com/library/newsletters/afterwork-sunset/email.html';
const MENU_URL = 'https://ysabelsociety.com/library/newsletters/afterwork-sunset/Ysabel_Fine_Dining_A5_Warm_White_Cover.pdf';
const SUBJECT = 'After work Sunset at Ysabel Society';

function createYsabelDraft() {
  const recipient = RECIPIENT.trim() || Session.getEffectiveUser().getEmail();
  validateRecipient_(recipient);
  const message = newsletterMessage_();
  const draft = GmailApp.createDraft(recipient, SUBJECT, message.body, {
    name: 'Ysabel Society',
    htmlBody: message.html
  });
  console.log('Draft created. Open Gmail > Drafts to review it.');
  console.log('Draft ID: ' + draft.getId());
}

function sendYsabelNewsletter() {
  const recipient = RECIPIENT.trim();
  validateRecipient_(recipient);
  const message = newsletterMessage_();
  GmailApp.sendEmail(recipient, SUBJECT, message.body, {
    name: 'Ysabel Society',
    htmlBody: message.html
  });
  console.log('Newsletter sent to ' + recipient);
}

function validateRecipient_(recipient) {
  if (!/^[^\s@,;]+@[^\s@,;]+\.[^\s@,;]+$/.test(recipient)) {
    throw new Error('Enter one email address in RECIPIENT at the top of this script.');
  }
}

function newsletterMessage_() {
  const response = UrlFetchApp.fetch(NEWSLETTER_URL, {muteHttpExceptions: true});
  if (response.getResponseCode() !== 200) {
    throw new Error('The public newsletter could not be loaded. HTTP ' + response.getResponseCode());
  }
  const html = response.getContentText('UTF-8');
  if (!html.includes('data-ysabel-newsletter="afterwork-sunset-email"')) {
    throw new Error('The address did not return the Ysabel email newsletter.');
  }
  return {
    html: html,
    body: "After work Sunset\nat Ysabel Society\n\nStarting 21 October. Wednesday to Friday, 16:00 - 19:00.\n\nTHE 80s & 90s RETURN\n\nSee you after work.\n\nThere's probably someone you've been meaning to see. A friend you haven't caught up with in a while, or a colleague you'd enjoy spending a little more time with outside the office.\n\nWe'd love to welcome you both to Ysabel Garden.\n\nFind something you like on the menu, enjoy a few drinks, and spend the sunset together with familiar songs from the 80s and 90s. The kind of music that brings back old memories and gives everyone something to talk about.\n\nWhether you're stopping by for an hour or staying into the evening, there's always room for a little more time together.\n\nExplore our Garden menu. We'd love to see you after work.\n\nView the menu: https://ysabelsociety.com/library/newsletters/afterwork-sunset/Ysabel_Fine_Dining_A5_Warm_White_Cover.pdf\nReserve your evening: https://www.ysabelsociety.com/\n"
  };
}
